cmake_minimum_required(VERSION 3.20)
project(SoD2Mcm LANGUAGES C CXX)
add_executable(NativeSettingsTest NativeSettingsTest.cpp)
target_compile_features(NativeSettingsTest PRIVATE cxx_std_17)
target_compile_options(NativeSettingsTest PRIVATE /utf-8 /W4 /WX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_subdirectory(vendor/minhook)
add_executable(NativeSettingsIggyTest NativeSettingsIggyTest.cpp)
target_compile_features(NativeSettingsIggyTest PRIVATE cxx_std_17)
target_compile_options(NativeSettingsIggyTest PRIVATE /utf-8 /W4 /WX)
target_compile_definitions(NativeSettingsIggyTest PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_include_directories(NativeSettingsIggyTest PRIVATE vendor/minhook/include)
target_link_libraries(NativeSettingsIggyTest PRIVATE minhook)
add_library(SoD2MeleeNative SHARED MeleeNative.cpp)
target_include_directories(SoD2MeleeNative PRIVATE vendor/minhook/include)
target_compile_features(SoD2MeleeNative PRIVATE cxx_std_17)
target_compile_definitions(SoD2MeleeNative PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_compile_options(SoD2MeleeNative PRIVATE /utf-8 /W4 /WX /EHa)
target_link_libraries(SoD2MeleeNative PRIVATE minhook)
set_target_properties(SoD2MeleeNative PROPERTIES OUTPUT_NAME "SoD2SE.MeleeSpeed.Native")
add_executable(MeleeNativeTest MeleeTest.cpp)
target_include_directories(MeleeNativeTest PRIVATE vendor/minhook/include)
target_compile_features(MeleeNativeTest PRIVATE cxx_std_17)
target_compile_definitions(MeleeNativeTest PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_compile_options(MeleeNativeTest PRIVATE /utf-8 /W4 /WX /EHa)
target_link_libraries(MeleeNativeTest PRIVATE minhook)
set(IMGUI vendor/imgui)
add_library(SoD2McmNative SHARED McmNative.cpp
    ${IMGUI}/imgui.cpp ${IMGUI}/imgui_draw.cpp ${IMGUI}/imgui_tables.cpp ${IMGUI}/imgui_widgets.cpp
    ${IMGUI}/backends/imgui_impl_win32.cpp ${IMGUI}/backends/imgui_impl_dx11.cpp)
target_include_directories(SoD2McmNative PRIVATE ${IMGUI} ${IMGUI}/backends vendor/minhook/include)
target_compile_features(SoD2McmNative PRIVATE cxx_std_17)
target_compile_definitions(SoD2McmNative PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_compile_options(SoD2McmNative PRIVATE /utf-8 /W4)
target_link_libraries(SoD2McmNative PRIVATE minhook d3d11 dxgi d3dcompiler user32)
set_target_properties(SoD2McmNative PROPERTIES OUTPUT_NAME "SoD2SE.Mcm.Native")

add_executable(McmRenderTest WIN32 RenderTest.cpp)

target_compile_features(McmRenderTest PRIVATE cxx_std_17)
target_compile_options(McmRenderTest PRIVATE /utf-8 /W4)
target_link_libraries(McmRenderTest PRIVATE d3d11 dxgi user32 shell32)
